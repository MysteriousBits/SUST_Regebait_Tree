\codeWithDescription{CRT}{nt/crt.cpp}
[Computes the unique solution $x$ to a system of congruences $x \equiv a_i \pmod{m_i}$ using the Chinese Remainder Theorem. Assumes all moduli are pairwise coprime and the answer fits in \texttt{ll}. \\
\textbf{Usage:} Store congruences as \texttt{\{remainder, modulus\}} pairs in \texttt{mods}. Call \texttt{crt(mods)} to obtain the smallest non-negative solution modulo the product of all moduli. Requires \texttt{extended\_gcd()} and \texttt{modinv()}. \\
\textbf{Time:} {$O(k \log M)$}]

\codeWithDescription{Mod Square Root}{nt/modsqrt.cpp}
[
  Tonelli-Shanks algorithm for modular square roots. Finds $x$ s.t. $x^2 = a \pmod p$ ($-x$ gives the other solution).
  **Time:** O(\log^2 p) worst case, O(\log p) for most $p$
]

\codeWithDescription{Discrete Logarithm}{nt/modlog.cpp}[%
  Solves $a^x \equiv b \pmod m$ for the smallest $x > 0$, using the Baby-Step Giant-Step algorithm.
  Supports non-coprime $a$ and $m$ by explicitly checking small exponents $x \le \sqrt{m}$ before applying step shifts.
  Can also compute the multiplicative order of $a$ via \texttt{modLog(a, 1, m)}.
][][modLog(a, b, m) // returns x > 0 or -1]

\codeWithDescription{Floor Sum}{nt/floor_sum_AP.cpp}
[
  Calculates $\sum_{i=0}^{n-1} \left\lfloor \frac{a \cdot i + b}{m} \right\rfloor$ in $O(\log m)$ time
  Handles negative $a, b$ and values $\ge m$ by automatically normalizing parameters modulo $m$.
  \textbf{Usage:} \texttt{fsum(n, m, a, b)}.
  \textbf{Time:} $O(\log m)$
]
